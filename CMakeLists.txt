cmake_minimum_required(VERSION 3.22)
project(Aether64 VERSION 1.0.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(aether_core STATIC
    src/core/math.cpp src/core/runtime.cpp src/core/launcher.cpp
    src/adapters/diagnostic.cpp)
target_include_directories(aether_core PUBLIC include)
set_target_properties(aether_core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
    target_compile_options(aether_core PRIVATE /W4)
else()
    target_compile_options(aether_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(ANDROID)
    add_subdirectory(src/android)
else()
    enable_testing()
    add_executable(aether_tests tests/core_tests.cpp)
    target_link_libraries(aether_tests PRIVATE aether_core)
    add_test(NAME core_contracts COMMAND aether_tests)
endif()
